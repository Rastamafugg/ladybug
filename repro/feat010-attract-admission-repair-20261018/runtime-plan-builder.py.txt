def sha(data: bytes) -> str:
    return hashlib.sha256(data).hexdigest()


def require(ok: bool, message: str) -> None:
    if not ok:
        raise ValueError(message)


def read(path: Path) -> bytes:
    require(path.is_file(), f"required A1-P artifact is missing: {path}")
    return path.read_bytes()


def require_digest(path: Path, expected: str, label: str) -> bytes:
    data = read(path)
    require(sha(data) == expected, f"{label} SHA-256 differs from the approved A1-P result")
    return data


def listing_row(rows, address: int, statement: str) -> bytes:
    matches = [code for pc, code, text in rows if pc == address and text == statement.lower()]
    require(len(matches) == 1, f"compiled listing lacks unique {statement} at ${address:04X}")
    return matches[0]


def branch_target(address: int, code: bytes, opcode: int) -> int:
    require(len(code) == 2 and code[0] == opcode, f"unexpected branch bytes at ${address:04X}")
    displacement = code[1] - 0x100 if code[1] & 0x80 else code[1]
    return address + 2 + displacement
