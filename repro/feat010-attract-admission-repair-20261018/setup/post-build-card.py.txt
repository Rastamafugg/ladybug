from __future__ import annotations

import hashlib
import importlib.util
import json
import re
import time
from pathlib import Path


WORK = Path(__file__).resolve().parents[1]
SETUP = WORK / "setup"
CANDIDATE = WORK / "candidate"
BUILD = CANDIDATE / "build"
STARTED = time.monotonic()
LIMIT_SECONDS = 60
HELPER_BASE = 0x06B2
PROFILE = {
    "LADYBUG_PROFILE": "complete",
    "LADYBUG_INPUT": "keyboard",
    "LADYBUG_CPU": "6809",
    "LADYBUG_ADAPTIVE": "1",
}
sys_path = SETUP
import sys
sys.path.insert(0, str(sys_path))
from helper_byte_decoder import parse_listing, parse_symbols, verify_helper_contract


def sha(data: bytes) -> str:
    return hashlib.sha256(data).hexdigest()


def require(ok: bool, message: str) -> None:
    if not ok:
        raise AssertionError(message)


def source_inventory() -> dict[str, str]:
    result = {}
    for folder in ("src", "scripts", "assets", "tiled"):
        for path in (CANDIDATE / folder).rglob("*"):
            if path.is_file() and "__pycache__" not in path.parts and path.suffix != ".pyc":
                result[path.relative_to(CANDIDATE).as_posix()] = sha(path.read_bytes())
    return result


def wiki_inventory() -> dict[str, str]:
    return {
        path.relative_to(CANDIDATE / "wiki").as_posix(): sha(path.read_bytes())
        for path in (CANDIDATE / "wiki").rglob("*") if path.is_file()
    }


def expected_argv() -> list[str]:
    absolute = str(CANDIDATE.resolve())
    require(absolute.lower().startswith("e:\\"), f"A1-P candidate is not on E: drive: {absolute}")
    wsl_path = "/mnt/e/" + absolute[3:].replace("\\", "/")
    return [
        "wsl", "-d", "Ubuntu", "--cd", wsl_path, "--exec", "timeout", "--signal=TERM",
        "--kill-after=2s", "595s", "env",
        *(f"{name}={value}" for name, value in PROFILE.items()),
        "bash", "scripts/build.sh", "build",
    ]


def stock_provenance_validate(receipt: dict):
    module_path = CANDIDATE / "scripts/source_reference/provenance.py"
    spec = importlib.util.spec_from_file_location("a1_candidate_provenance", module_path)
    require(spec is not None and spec.loader is not None, "candidate stock provenance module is unavailable")
    module = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(module)
    config_path = BUILD / "source-reference-adaptive.json"
    config_bytes = config_path.read_bytes()
    require(receipt.get("artifacts", {}).get(config_path.name) == sha(config_bytes),
            "generated adaptive source-reference config is not bound by the build receipt")
    config = json.loads(config_bytes)
    validated = module.validate(CANDIDATE, BUILD, config)
    require(validated == receipt, "stock provenance validator returned a different receipt")
    return module_path, sha(config_bytes)


def main() -> None:
    baseline = json.loads((SETUP / "baseline-inputs.json").read_text(encoding="utf-8"))
    projection = json.loads((SETUP / "projected-inputs.json").read_text(encoding="utf-8"))
    wiki_manifest = json.loads((SETUP / "wiki-inputs.json").read_text(encoding="utf-8"))
    source_result = json.loads((SETUP / "source-card-result.json").read_text(encoding="utf-8"))
    attempt_path, result_path, log_path = (SETUP / "build-attempt.json", SETUP / "build-result.json",
                                           SETUP / "build-once.log")
    attempt_bytes, result_bytes, log_bytes = attempt_path.read_bytes(), result_path.read_bytes(), log_path.read_bytes()
    attempt, build_result = json.loads(attempt_bytes), json.loads(result_bytes)
    source_card_path = SETUP / "source-card.py.txt"
    decoder_path = SETUP / "helper_byte_decoder.py"
    build_wrapper_path = SETUP / "build-once.py.txt"
    source_card_hash, decoder_hash = sha(source_card_path.read_bytes()), sha(decoder_path.read_bytes())
    require(source_result.get("card") == "A1-S" and source_result.get("outcome") == "PASS"
            and source_result.get("source_card_sha256") == source_card_hash
            and source_result.get("decoder_sha256") == decoder_hash,
            "passing A1-S result does not bind the current source card and shared decoder")
    require(projection.get("projection_only") is True and projection.get("input_count") == 266
            and projection.get("changed_inputs") == ["src/perimeter_reset_helper.s"]
            and baseline.get("input_count") == 266 and wiki_manifest.get("file_count") == 379,
            "A1-P projected source/wiki manifests differ from the approved closure")

    argv = expected_argv()
    wrapper_hash = sha(build_wrapper_path.read_bytes())
    wiki = wiki_inventory()
    wiki_hash = sha(json.dumps(wiki, sort_keys=True, separators=(",", ":")).encode("utf-8"))
    expected_inputs = dict(baseline["inputs"])
    expected_inputs["src/perimeter_reset_helper.s"] = projection["inputs"]["src/perimeter_reset_helper.s"]
    inputs = source_inventory()
    require(inputs == projection["inputs"] == expected_inputs and len(inputs) == 266,
            "candidate source closure differs from the projected one-file source manifest")
    require(wiki == wiki_manifest.get("files") and len(wiki) == 379 and wiki_hash == wiki_manifest.get("inventory_sha256"),
            "candidate wiki differs from the frozen 379-file inventory")
    require(not (CANDIDATE / ".git").exists(), "candidate unexpectedly contains Git state")

    require(attempt.get("card") == "A1-B" and attempt.get("attempt") == 1
            and attempt.get("candidate") == "repro/feat010-attract-admission-repair-20261018/candidate"
            and attempt.get("baseline_rom_sha256") == "6b2181bfba91869fcfe304524ecdd965d30c7b5ad6c560c0b00da67439282b08",
            "A1-B attempt marker is not the approved single build")
    require(attempt.get("wrapper_sha256") == wrapper_hash == build_result.get("wrapper_sha256")
            and attempt.get("argv") == argv == build_result.get("argv")
            and attempt.get("profile") == PROFILE == build_result.get("profile")
            and attempt.get("source_card_sha256") == source_card_hash == build_result.get("source_card_sha256")
            and attempt.get("decoder_sha256") == decoder_hash == build_result.get("decoder_sha256"),
            "A1-B wrapper identity, exact argv, or complete/keyboard/6809/adaptive profile differs")
    require(attempt.get("source_input_count") == len(inputs) == build_result.get("source_input_count")
            and attempt.get("projected_helper_source_sha256") == inputs["src/perimeter_reset_helper.s"]
            and attempt.get("wiki_inventory_sha256") == wiki_hash == build_result.get("wiki_inventory_sha256"),
            "A1-B attempt is not bound to the approved candidate inputs/wiki")
    require(build_result.get("card") == "A1-B" and build_result.get("attempt") == 1
            and build_result.get("outcome") == "BUILD RETURNED ZERO; A1-P PENDING"
            and build_result.get("returncode") == 0 and build_result.get("timed_out") is False
            and build_result.get("elapsed_seconds", LIMIT_SECONDS + 1) <= 600
            and build_result.get("whole_phase_limit_seconds") == 600
            and build_result.get("child_limit_seconds") == 595 and build_result.get("kill_grace_seconds") == 2,
            "A1-B did not complete its one bounded build phase successfully")
    require(attempt.get("result_sha256") == sha(result_bytes)
            and build_result.get("log") == "build-once.log"
            and build_result.get("log_sha256") == sha(log_bytes),
            "A1-B result/log hash chain is incomplete")

    receipt_path = BUILD / "source-build-receipt.json"
    receipt_bytes = receipt_path.read_bytes()
    receipt = json.loads(receipt_bytes)
    require(build_result.get("receipt_sha256") == sha(receipt_bytes),
            "A1-B receipt hash differs from the actual source-build receipt")
    require(receipt.get("state") == "complete" and receipt.get("profile") == "complete"
            and receipt.get("inputs") == inputs,
            "compiled receipt is incomplete, wrong-profile, or stale against candidate sources")
    provenance_path, provenance_config_hash = stock_provenance_validate(receipt)
    require("-DADAPTIVE_RENDERING=1" in [argument for invocation in receipt.get("invocations", [])
                                        for argument in invocation.get("arguments", [])],
            "receipt does not include the approved adaptive-rendering assembler definition")

    rom_path = BUILD / "ladybug.rom"
    rom = rom_path.read_bytes()
    artifacts = receipt.get("artifacts", {})
    require(len(rom) == 65536 and artifacts.get("ladybug.rom") == sha(rom)
            and build_result.get("rom_sha256") == sha(rom),
            "packaged GMC ROM identity or 64-KiB format differs from the A1-B receipt")
    helper_path = BUILD / "ladybug-perimeter-reset-helper.bin"
    map_path = BUILD / "ladybug-perimeter-reset-helper.map"
    listing_path = BUILD / "ladybug-perimeter-reset-helper.lst"
    helper, map_bytes, listing_bytes = helper_path.read_bytes(), map_path.read_bytes(), listing_path.read_bytes()
    require(artifacts.get(helper_path.name) == sha(helper) and artifacts.get(map_path.name) == sha(map_bytes)
            and artifacts.get(listing_path.name) == sha(listing_bytes),
            "compiled helper binary/map/listing do not match receipt-bound hashes")
    symbols, listing_rows = parse_symbols(map_bytes.decode("utf-8")), parse_listing(listing_bytes.decode("utf-8"))
    decoded = verify_helper_contract(helper, symbols, listing_rows)
    require(decoded["helper_bytes"] == 334 and decoded["helper_end"] == HELPER_BASE + 334,
            "compiled helper does not fill the approved 334/334-byte allocation")

    presentation_path = BUILD / "ladybug-presentation-runtime.bin"
    presentation = presentation_path.read_bytes()
    require(len(presentation) == 1280 and artifacts.get(presentation_path.name) == sha(presentation),
            "full presentation module changed from its receipt-bound 1,280-byte cap")
    main_map = (BUILD / "ladybug.map").read_text(encoding="utf-8")
    main_symbols = parse_symbols(main_map)
    resident_end, asset_start, asset_end = (main_symbols[name] for name in ("resident_end", "asset_start", "asset_end"))
    resident_used, asset_used = resident_end - 0xC000, asset_end - asset_start
    require(asset_start == 0xE000 and resident_used <= 8160 and 8192 - resident_used >= 32
            and asset_used <= 7680, "resident minimum free headroom or asset cap failed")
    log_text = log_bytes.decode("utf-8", errors="replace")
    capacity = [tuple(map(int, row)) for row in re.findall(
        r"^build: resident (\d+)/8192 bytes; assets (\d+)/7680 bytes$", log_text, re.M)]
    require(capacity and capacity[-1] == (resident_used, asset_used)
            and all(resident <= 8160 and assets <= 7680 for resident, assets in capacity),
            "ordinary build capacity-summary records differ from map or include a cap miss")
    require(time.monotonic() - STARTED <= LIMIT_SECONDS, "A1-P exceeded its whole-phase 60-second limit")

    result = {
        "card": "A1-P", "outcome": "PASS", "receipt_sha256": sha(receipt_bytes),
        "rom_sha256": sha(rom), "helper_sha256": sha(helper), "helper_map_sha256": sha(map_bytes),
        "helper_listing_sha256": sha(listing_bytes),
        "helper_source_sha256": inputs["src/perimeter_reset_helper.s"],
        "profile": build_result["profile"], "source_inputs": len(inputs), "wiki_files": len(wiki),
        "helper_bytes_of_cap": f"{len(helper)}/334", "helper_address_range": f"${HELPER_BASE:04X}-${decoded['helper_end'] - 1:04X}",
        "helper_exclusive_end": f"${decoded['helper_end']:04X}",
        "presentation_bytes_of_cap": f"{len(presentation)}/1280",
        "decoder": {key: f"${value:04X}" if isinstance(value, int) else value
                    for key, value in decoded.items()},
        "resident_used_of_8192": resident_used, "resident_free_minimum": 8192 - resident_used,
        "asset_used_of_7680": asset_used, "ordinary_capacity_rows": len(capacity),
        "stock_provenance_validation": "PASS", "stock_provenance_module_sha256": sha(provenance_path.read_bytes()),
        "generated_provenance_config_sha256": provenance_config_hash,
        "duration_seconds": round(time.monotonic() - STARTED, 3), "whole_phase_limit_seconds": LIMIT_SECONDS,
    }
    (SETUP / "post-build-result.json").write_text(json.dumps(result, indent=2) + "\n",
                                                  encoding="utf-8", newline="\n")
    print(json.dumps(result, indent=2))


if __name__ == "__main__":
    main()
