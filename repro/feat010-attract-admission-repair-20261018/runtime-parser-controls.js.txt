// Offline controls for the A1-R parser. The runtime controller uses the same rules.
function readMemory(output, start, length) {
  const found = [];
  for (const line of output.split(/\r?\n/)) {
    const match = line.match(/^\s*0x([0-9a-f]+)(?:\s+<[^>]*>)?:\s*(.*)$/i);
    if (!match) continue;
    if (parseInt(match[1], 16) !== start + found.length) throw new Error('noncontiguous memory');
    found.push(...[...match[2].matchAll(/\b0x([0-9a-f]{2})\b/gi)].map(m => parseInt(m[1], 16)));
  }
  if (found.length !== length) throw new Error('wrong memory length');
  return found;
}
function readState(bytes, fields) {
  const values = {};
  for (const field of fields) {
    const match = field.match(/^(.*) ([0-9a-f]{4})(?:-([0-9a-f]{4}))?$/i);
    if (!match) throw new Error('malformed state field');
    const first = parseInt(match[2], 16), last = parseInt(match[3] ?? match[2], 16);
    if (first < 2 || last > 0xe8 || last < first) throw new Error('state outside approved span');
    let value = 0;
    for (let address = first; address <= last; address++) value = value * 256 + bytes[address - 2];
    values[match[1]] = value;
  }
  return values;
}
function readPcDp(output) {
  const match = output.match(/stop_pc=([0-9a-f]+)\s+stop_dp=([0-9a-f]+)/i);
  if (!match) throw new Error('missing raw PC/DP');
  return {pc: parseInt(match[1], 16), dp: parseInt(match[2], 16)};
}
function readRegister(output, name) {
  const line = output.split(/\r?\n/).find(row => new RegExp('^\\s*' + name + '(?:\\s|:|=)', 'i').test(row));
  if (!line) throw new Error('missing register ' + name);
  const tail = line.replace(new RegExp('^\\s*' + name + '\\s*(?:[:=]\\s*)?', 'i'), '');
  const match = tail.match(/(?:0x)?([0-9a-f]+)/i);
  if (!match) throw new Error('unparseable register ' + name);
  return parseInt(match[1], 16);
}
function checkSpan(output, start, expectedHex) {
  const actual = readMemory(output, start, expectedHex.length / 2)
    .map(value => value.toString(16).padStart(2, '0')).join('');
  if (actual !== expectedHex) throw new Error('live bytes differ');
  return actual;
}
function phaseForTimer(timer) {
  return timer < 6 ? 0 : ((((timer - 6) & 0xff) >> 3) & 3);
}
function stateBytes(values) {
  const bytes = new Uint8Array(231);
  const set = (address, value, length = 1) => {
    for (let offset = 0; offset < length; offset++) {
      bytes[address - 2 + length - offset - 1] = value & 0xff;
      value >>>= 8;
    }
  };
  set(0x0091, values.PENDING ?? 0);
  set(0x00b0, values.PRES_TIMER ?? 0, 2);
  set(0x00d3, values.PRES_ACTOR_PHASE ?? 0);
  return bytes;
}
function dump(bytes, start = 2) {
  const lines = [];
  for (let offset = 0; offset < bytes.length; offset += 8) {
    const chunk = [...bytes.slice(offset, offset + 8)].map(value => '0x' + value.toString(16).padStart(2, '0')).join(' ');
    lines.push('0x' + (start + offset).toString(16).padStart(4, '0') + ': ' + chunk);
  }
  return lines.join('\n');
}
function checkFirstStop(codeOutput, registerOutput, rawOutput, p0Fields) {
  const bytes = readMemory(codeOutput, 0x078c, 4);
  const actual = bytes.map(value => value.toString(16).padStart(2, '0')).join('');
  if (actual !== '0d912653') throw new Error('guard span mismatch');
  const regs = readPcDp(rawOutput);
  if (regs.pc !== 0x078e || regs.dp !== 0) throw new Error('wrong first stop PC/DP');
  if (readRegister(registerOutput, 'pc') !== regs.pc || readRegister(registerOutput, 'dp') !== regs.dp) throw new Error('register and raw PC/DP differ');
  const cc = readRegister(registerOutput, 'cc');
  const displacement = bytes[3] & 0x80 ? bytes[3] - 0x100 : bytes[3];
  if (0x078e + 2 + displacement !== 0x07e3) throw new Error('BNE target differs from pao_done');
  if ((cc & 0x04) !== 0) return {coverage: 'missing-positive', p0: p0Fields.PRES_ACTOR_PHASE, cc};
  return {coverage: 'qualified', p0: p0Fields.PRES_ACTOR_PHASE, cc};
}
function checkSecondStop(codeOutput, rawOutput, fields, p0) {
  checkSpan(codeOutput, 0x07e3, '39');
  const regs = readPcDp(rawOutput);
  if (regs.pc !== 0x07e3 || regs.dp !== 0) throw new Error('wrong pao_done stop PC/DP');
  if (fields.PRES_ACTOR_PHASE !== p0) throw new Error('actor phase cache changed on guarded path');
  return true;
}
function checkThirdStop(codeOutput, rawOutput, fields) {
  checkSpan(codeOutput, 0x0792, 'bd0ddf');
  const regs = readPcDp(rawOutput);
  if (regs.pc !== 0x0792 || regs.dp !== 0) throw new Error('wrong prepare-entry PC/DP');
  if (fields.PENDING !== 0) throw new Error('PENDING is not clear before prepare');
  if (fields.PRES_ACTOR_PHASE !== phaseForTimer(fields.PRES_TIMER)) throw new Error('phase differs from timer function');
  return true;
}
