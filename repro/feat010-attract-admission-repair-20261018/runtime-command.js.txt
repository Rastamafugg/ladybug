function fail(message, kind = 'OBSERVER FAILURE; STOP') {
  const error = new Error(message); error.kind = kind; throw error;
}
function requireReady() {

  if (!ready || !ready.parent_review_ready || ready.plan_sha256 !== plan.plan_sha256) fail('Parent Ready metadata does not bind the frozen phase-revision plan');

  const planHashes = plan.gdb_tool_control_hashes ?? {};
  const hashNames = Object.keys(planHashes).sort();
  if (JSON.stringify(Object.keys(readyHashes).sort()) !== JSON.stringify(hashNames)
      || hashNames.some(name => readyHashes[name] !== planHashes[name])) fail('Current GDB tool hashes differ from the frozen plan');
  if (plan.target_path_qualified !== false || plan.artifacts.rom_sha256 !== '2dcb1d4e8c1d08903e7474ede6e4051ecf211a31512d8e42d0a8c5fb86df7c2b'
      || plan.artifacts.presentation_source_sha256 !== '6c7525fbc98e349128e58310e805aca6fd8b3a209f252445dba9e7bfcad85123'
      || plan.artifacts.presentation_sha256 !== '53720dd325f3db6c74259d67a987b672f2f3b277a0313ac0214073a8a9da04f2'
      || plan.artifacts.presentation_map_sha256 !== '7a52ebeb75c22f146e6ebb6748388685b4c27773bf88cbcf7d7a1d871103991c'
      || plan.artifacts.presentation_listing_sha256 !== '7c2c7f93e3531c233ff0c1d1b3e0e197d0d10d833abd2d807178228975531329') fail('Plan is not bound to the current receipt-qualified A1-P presentation artifact');
}
function outputs(result) {
  if (result.isError) { audit.last_tool_error = result; fail('GDB-MCP call returned an error'); }
  const rows = result.structuredContent?.result ?? result.content.filter(c => c.type === 'text').map(c => JSON.parse(c.text));
  if (!Array.isArray(rows) || rows.some(row => row.error)) { audit.last_tool_error = result; fail('GDB command failed; no retry'); }
  return rows;
}
async function batch(commands, timeout) {
  const elapsed = (Date.now() - started) / 1000;
  const commandBudget = commands.length * timeout;
  const recovery = plan.execution_recipe.prompt_recovery_reserve_seconds;
  const cleanup = plan.execution_recipe.cleanup_reserve_seconds;
  const overhead = plan.execution_recipe.call_local_overhead_reserve_seconds;
  const admittedSeconds = commandBudget + recovery + cleanup + overhead;
  if (elapsed + admittedSeconds > plan.execution_recipe.whole_phase_deadline_seconds
      || elapsed + commandBudget + recovery + overhead > plan.execution_recipe.cleanup_begins_by_seconds) {
    fail('Insufficient remaining deadline for this GDB call and cleanup reserve', 'INCOMPLETE DEADLINE BOUNDARY; STOP');
  }
  const before = Date.now();
  const result = await tools.mcp__gdb_mcp__batch_commands({session_id: session, commands, timeout, stop_on_error: true});
  const rows = outputs(result);
  audit.gdb_calls.push({commands, timeout_seconds_per_command: timeout,
                        command_timeout_budget_seconds: commandBudget,
                        elapsed_ms: Date.now() - before,
                        outputs: rows.map(row => ({command: row.command, output: row.output}))});
  return rows;
}
function memory(output, start, length) {
  const found = [];
  for (const line of output.split(/\r?\n/)) {
    const match = line.match(/^\s*0x([0-9a-f]+)(?:\s+<[^>]*>)?:\s*(.*)$/i);
    if (!match) continue;
    if (parseInt(match[1], 16) !== start + found.length) fail('Noncontiguous live memory dump');
    found.push(...[...match[2].matchAll(/\b0x([0-9a-f]{2})\b/gi)].map(m => parseInt(m[1], 16)));
  }
  if (found.length !== length) fail('Live memory dump length differs from the approved span');
  return found;
}
function stateFields(bytes) {
  const values = {};
  for (const field of plan.state_read.report_fields) {
    const match = field.match(/^(.*) ([0-9a-f]{4})(?:-([0-9a-f]{4}))?$/i);
    if (!match) fail('Malformed frozen state field');
    const first = parseInt(match[2], 16), last = parseInt(match[3] ?? match[2], 16);
    if (first < 2 || last > 0xe8 || last < first) fail('State field lies outside the approved RAM dump');
    let value = 0;
    for (let address = first; address <= last; address++) value = value * 256 + bytes[address - 2];
    values[match[1]] = value;
  }
  return values;
}
function pcDp(output) {
  const match = output.match(/stop_pc=([0-9a-f]+)\s+stop_dp=([0-9a-f]+)/i);
  if (!match) fail('Raw stopped PC/DP print is missing');
  return {pc: parseInt(match[1], 16), dp: parseInt(match[2], 16)};
}
function registerValue(output, name) {
  const escaped = name.replace(/[.*+?^${}()|[\]\\]/g, '\\$&');
  const line = output.split(/\r?\n/).find(row => new RegExp('^\\s*' + escaped + '(?:\\s|:|=)', 'i').test(row));
  if (!line) fail('`info registers pc dp cc` omitted register ' + name);
  const tail = line.replace(new RegExp('^\\s*' + escaped + '\\s*(?:[:=]\\s*)?', 'i'), '');
  const match = tail.match(/(?:0x)?([0-9a-f]+)/i);
  if (!match) fail('Cannot parse register ' + name + ' from raw GDB output');
  return parseInt(match[1], 16);
}
function qualifySpan(stop, rows) {
  const span = plan.code_spans.find(item => item.id === stop.span_id);
  const row = rows.find(item => item.command === stop.code_command);
  if (!span || !row) fail('Frozen live span or matching GDB response is missing');
  const start = parseInt(span.address, 16);
  const actual = memory(row.output, start, span.length).map(value => value.toString(16).padStart(2, '0')).join('');
  if (actual !== span.expected_hex) {
    audit.failure_span = {id: span.id, address: span.address, expected_hex: span.expected_hex, actual_hex: actual};
    fail('Live helper bytes differ from receipt-bound source/staged bytes', 'BYTE IDENTITY MISMATCH; STOP');
  }
  return {span, actual};
}
function phaseFromTimer(timer) {
  return timer < 6 ? 0 : ((((timer - 6) & 0xff) >> 3) & 3);
}
