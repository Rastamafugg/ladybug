def require(value, message):
    if not value:
        raise AssertionError(message)


def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def expected_rejection(call, label):
    try:
        call()
    except (ValueError, OSError, KeyError, AssertionError):
        result["controls"].append({"name": label, "result": "expected rejection"})
        return
    raise AssertionError(label + " unexpectedly accepted")


def guarded(tree, diagnostic):
    hits = [
        node for node in ast.walk(tree)
        if isinstance(node, ast.If)
        and len(node.body) == 1
        and isinstance(node.body[0], ast.Raise)
        and any(
            isinstance(value, ast.Constant) and value.value == diagnostic
            for value in ast.walk(node.body[0])
        )
    ]
    require(len(hits) == 1, "unique maintained assertion: " + diagnostic)
    return hits[0]


def masked_ast(tree, nodes):
    clone = copy.deepcopy(tree)
    keys = {(node.lineno, node.col_offset) for node in nodes}
    found = 0
    for node in ast.walk(clone):
        if isinstance(node, ast.If) and (node.lineno, node.col_offset) in keys:
            node.test = ast.Constant(value="APPROVED_PREDICATE_MASK")
            found += 1
    require(found == len(keys), "AST mask node count")
    return ast.dump(clone)


def apply_exact_patch(original, patch_bytes):
    patch_lines = patch_bytes.splitlines(keepends=True)
    hunks = []
    index = 0
    while index < len(patch_lines):
        if patch_lines[index].startswith(b"@@"):
            index += 1
            old_lines = []
            new_lines = []
            while index < len(patch_lines) and not patch_lines[index].startswith(
                (b"@@", b"--- ", b"+++ ")
            ):
                line = patch_lines[index]
                if line.startswith(b"-"):
                    old_lines.append(line[1:].rstrip(b"\r\n").decode("utf-8"))
                elif line.startswith(b"+"):
                    new_lines.append(line[1:].rstrip(b"\r\n").decode("utf-8"))
                index += 1
            if old_lines or new_lines:
                hunks.append((old_lines, new_lines))
        else:
            index += 1
    require(len(hunks) == 2 and all(len(a) == len(b) == 1 for a, b in hunks),
            "maintained patch must contain exactly two one-line predicate changes")
    rows = original.decode("utf-8").splitlines(keepends=True)
    for old_lines, new_lines in hunks:
        matches = [
            i for i, row in enumerate(rows)
            if row.rstrip("\r\n") == old_lines[0]
        ]
        require(len(matches) == 1, "unique maintained patch source line")
        i = matches[0]
        ending = "\r\n" if rows[i].endswith("\r\n") else "\n" if rows[i].endswith("\n") else ""
        rows[i] = new_lines[0] + ending
    return "".join(rows).encode("utf-8")
