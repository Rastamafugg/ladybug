def timing_contract(src):
    flat=re.sub(r'\s+','',src)
    if 'asyncfunctionresumeExpected(expectedPC,waitSeconds=5)' not in flat:return 'non-input wait default changed'
    calls=re.findall(r'awaitresumeExpected\(([^)]*)\)',flat)
    if calls!=['0xc116','0x1900','0x1a78,15','0x1ace','0x1b48','0x1bc1']:return 'selected-wait scope mismatch'
    for required in ['requireTime(waitSeconds*1000+5000)','timeout:waitSeconds','if(inputStageRemainingMs<37000)','return45000-(Date.now()-started)','remaining()<ms+8000','++observations>12']:
        if required not in flat:return 'missing timing guard: '+required
    arm=src.index('await cmd("tbreak *0x1a78 if $a == 7")')
    guard=src.index('if(inputStageRemainingMs<37000)')
    notice=src.index('notify("Credits input anchor')
    run=src.index('await resumeExpected(0x1a78,15)')
    if not arm<guard<notice<run:return 'input notification order'
    return 'PASS'
