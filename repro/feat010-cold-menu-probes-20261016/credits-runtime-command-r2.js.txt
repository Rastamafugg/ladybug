async function host(py, cleanup=false) {

async function resumeExpected(expectedPC,waitSeconds=5) {
  requireTime(waitSeconds*1000+5000);
  if (++observations > 12) throw Error("Observation cap");
  const continueStartedAt=Date.now();
  let result, commandError;
  try { result=await tools.mcp__gdb_mcp__continue_exec({session_id:sid,timeout:waitSeconds}); } catch(e) { commandError=e; }
  const continueStoppedAt=Date.now();
  trace.push({at_ms:continueStoppedAt-started,continue_start_utc:new Date(continueStartedAt).toISOString(),continue_stop_utc:new Date(continueStoppedAt).toISOString(),continue_elapsed_ms:continueStoppedAt-continueStartedAt,continue_result:result,expectedPC,expected_wait_seconds:waitSeconds,scheduled_host_interrupt:false});
  if (commandError || !result || result.isError || !/breakpoint-hit/i.test(String(body(result)))) {
    await boundedTimeoutSnapshot("expected-pc-"+expectedPC.toString(16));
    throw Error(commandError ? String(commandError) : "Expected natural breakpoint not reported within "+waitSeconds+" seconds");
  }
  const sync=await cmd("echo COLD_MENU_STOP_SYNC");
  if (!sync.includes("COLD_MENU_STOP_SYNC")) throw Error("Observer stop response not synchronized");
  const raw=await cmd("p/x $pc");
  const match=raw.match(/=\s*0x([0-9a-f]+)/i);
  if (!match) throw Error("PC response not understood: "+raw);
  const pc=parseInt(match[1],16);
  if (pc!==expectedPC) {
    await snapshot("unexpected-"+expectedPC.toString(16),false);
    throw Error("First unexpected stop: "+pc.toString(16));
  }
  return pc;
}

  requireTime(14000);
  if (++observations > 12) throw Error("Observation cap");

  trace.push({at_ms:continueStoppedAt-started,continue_start_utc:new Date(continueStartedAt).toISOString(),continue_stop_utc:new Date(continueStoppedAt).toISOString(),continue_elapsed_ms:continueStoppedAt-continueStartedAt,continue_result:result,expectedPC:null,expected_wait_seconds:5,final_witness:true});
