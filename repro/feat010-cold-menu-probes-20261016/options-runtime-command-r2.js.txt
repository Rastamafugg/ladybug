async function resumeExpected(expectedPC) {

  });

  if(commandError) throw commandError;
  if(controlError || result.isError) throw Error(controlError || "Final witness continue failed/timeout");
  const stopBody=String(body(result));
  const signalStop=/signal-received|SIGINT/i.test(stopBody);
  if(fired && !signalStop) throw Error("Timer/natural-stop race; observer state incomplete");
  const sync=await cmd("echo COLD_MENU_STOP_SYNC");
  if(!sync.includes("COLD_MENU_STOP_SYNC")) throw Error("Observer stop response not synchronized");
  const raw=await cmd("p/x $pc");
  const match=raw.match(/=\s*0x([0-9a-f]+)/i);
  if(!match) throw Error("PC response not understood: "+raw);
  const pc=parseInt(match[1],16);
  if(!fired || !signalStop) {
    await snapshot("unexpected",false);
    throw Error("Final witness did not end at the owned timer signal: "+pc.toString(16));
  }
  return pc;
}
