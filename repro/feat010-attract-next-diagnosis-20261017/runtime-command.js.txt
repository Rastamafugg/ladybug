function fail(message, kind = 'OBSERVER FAILURE; STOP') {
  const error = new Error(message); error.kind = kind; throw error;
}
function outputs(result) {
  if (result.isError) { audit.failure_tool_result = result; fail('GDB tool error'); }
  const rows = result.structuredContent?.result ?? result.content.filter(c => c.type === 'text').map(c => JSON.parse(c.text));
  if (!Array.isArray(rows) || rows.some(r => r.error)) { audit.failure_tool_result = result; fail('GDB command error'); }
  return rows;
}
function memory(output, start, length) {
  const found = [];
  for (const line of output.split(/\r?\n/)) {
    const match = line.match(/^\s*0x([0-9a-f]+)(?:\s+<[^>]*>)?:\s*(.*)$/i);
    if (!match) continue;
    if (parseInt(match[1], 16) !== start + found.length) fail('Noncontiguous memory dump');
    found.push(...[...match[2].matchAll(/\b0x([0-9a-f]{2})\b/gi)].map(m => parseInt(m[1], 16)));
  }
  if (found.length !== length) fail('Memory dump length differs from approved span');
  return found;
}
function fields(bytes) {
  const values = {};
  for (const field of plan.state_read.report_fields) {
    const match = field.match(/^(.*) ([0-9a-f]{4})(?:-([0-9a-f]{4}))?$/i);
    if (!match) fail('Malformed frozen state field');
    const first = parseInt(match[2], 16), last = parseInt(match[3] ?? match[2], 16);
    if (first < 2 || last > 0xe8 || last < first) fail('State field outside approved RAM span');
    let value = 0;
    for (let address = first; address <= last; address++) value = value * 256 + bytes[address - 2];
    values[match[1]] = value;
  }
  if (![0, 1].includes(values.FB_BACK_ID)) fail('Unexpected framebuffer ID');
  values.selected_BACK_logo_cache = values.FB_BACK_ID === 0 ? values['logo cache A'] : values['logo cache B'];
  return values;
}
