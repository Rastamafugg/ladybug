def sha(data: bytes) -> str:
    return hashlib.sha256(data).hexdigest()


def require(condition: bool, message: str) -> None:
    if not condition:
        raise AssertionError(message)


def symbol(text: str, name: str) -> int:
    matches = re.findall(rf"(?m)^Symbol: {re.escape(name)}\s+.* = ([0-9A-Fa-f]+)$", text)
    require(len(matches) == 1, f"map symbol {name} missing or ambiguous")
    return int(matches[0], 16)


def decode_schedule(data: bytes) -> list[tuple[int, int, int, int]]:
    require(len(data) % 7 == 0, f"schedule byte length {len(data)} is not a multiple of seven")
    return [
        (int.from_bytes(data[i:i + 2], "big"), int.from_bytes(data[i + 2:i + 4], "big"),
         int.from_bytes(data[i + 4:i + 6], "big"), data[i + 6])
        for i in range(0, len(data), 7)
    ]


def tmx_rank_rows(path: Path) -> list[int]:
    root = ET.parse(path).getroot()
    branding = None
    for layer in root.findall("layer"):
        if layer.get("name") != "High Score Table and Branding":
            continue
        width, height = int(layer.get("width")), int(layer.get("height"))
        flat = [int(value) for value in layer.find("data").text.replace("\n", "").split(",") if value.strip()]
        require(width == 40 and height == 24 and len(flat) == width * height, "frozen TMX layout identity/geometry mismatch")
        branding = [flat[y * width:(y + 1) * width] for y in range(height)]
    require(branding is not None, "TMX High Score branding layer missing")
    require([x for x, value in enumerate(branding[4]) if value] == [10, 11, 12], "authored TOP label is not row 4")
    require(not any(branding[5]), "authored separator row 5 is not blank")
    rows = [row for row in range(6, 14) if [x for x, value in enumerate(branding[row]) if value] == [10, 11, 12]]
    require(rows == list(range(6, 14)), "TMX ordinal labels do not occupy rows 6 through 13")
    return rows


def cell(destination: int) -> tuple[int, int]:
    offset = destination - 0x2000
    require(offset >= 0 and offset % 4 == 0, f"unaligned or non-screen destination {destination:04X}")
    row, row_offset = divmod(offset, 1280)
    return row, row_offset // 4


def expected_schedule(rank_rows: list[int]) -> list[tuple[int, int, int, int]]:
    records = []
    for index, row in enumerate(rank_rows):
        source = 0xAF8E + 10 * index
        records.extend(((320 + 64 * index, source, 0x2000 + row * 1280 + 64, 1),
                        (352 + 64 * index, source, 0x2000 + row * 1280 + 108, 0)))
    records.extend(((832, 0xAF84, 0x3440, 1), (864, 0xAF84, 0x346C, 0)))
    return records


def assert_no_separator_write(records: list[tuple[int, int, int, int]]) -> None:
    if any(cell(destination)[0] == 5 for _, _, destination, _ in records):
        raise AssertionError("compiled schedule writes into authored blank separator row 5")


def assert_good_schedule(records: list[tuple[int, int, int, int]], rank_rows: list[int]) -> None:
    assert_no_separator_write(records)
    require(len(records) == 18, f"compiled schedule has {len(records)} jobs, expected 18")
    require(records == expected_schedule(rank_rows), "compiled schedule changed a retained threshold/source/destination/flag")
    sources = [0xAF84, *(0xAF8E + 10 * i for i in range(8))]
    for source in sources:
        require(sum(item[1] == source and item[3] == 1 for item in records) == 1
                and sum(item[1] == source and item[3] == 0 for item in records) == 1,
                f"record {source:04X} lacks exactly one compiled name and score job")
    for threshold, source, destination, flag in records:
        row, column = cell(destination)
        width = 7 if flag else 6
        require(column == (16 if flag else 27) and row in (4, *rank_rows),
                f"compiled field address invalid for {source:04X}")
        require(threshold % 32 == 0 and threshold < 960
                and row * 40 + column + width - 1 < threshold,
                f"compiled full-field slice bound invalid at {column},{row}")
    require([item for item in records if item[1] == 0xAF84]
            == [(832, 0xAF84, 0x3440, 1), (864, 0xAF84, 0x346C, 0)],
            "TOP record zero is not retained exclusively at authored row 4")


def schedule_from_binary(binary: bytes, map_text: str) -> list[tuple[int, int, int, int]]:
    start = symbol(map_text, "highscore_phase_tick")
    end = symbol(map_text, "highscore_phase_helper_end")
    table_start = symbol(map_text, "ranking_schedule_data")
    table_end = symbol(map_text, "ranking_schedule_end")
    require(end - start == len(binary) <= 3136, "helper file size/map span exceeds the 3,136-byte cap")
    offset, length = table_start - start, table_end - table_start
    require(offset >= 0 and length > 0 and offset + length <= len(binary), "mapped schedule falls outside helper binary")
    return decode_schedule(binary[offset:offset + length])


def load_stock_provenance():
    sys.dont_write_bytecode = True
    source = CANDIDATE / "scripts/source_reference/provenance.py"
    spec = importlib.util.spec_from_file_location("feat010_sep_stock_provenance", source)
    require(spec is not None and spec.loader is not None, "candidate stock provenance module cannot be loaded")
    module = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(module)
    return module


def main() -> None:
    projected = json.loads((SETUP / "projected-inputs.json").read_text(encoding="utf-8"))
    wiki = json.loads((SETUP / "wiki-inputs.json").read_text(encoding="utf-8"))
    prep = json.loads((SETUP / "preparation.json").read_text(encoding="utf-8"))
    require(projected.get("projection_only") and projected.get("input_count") == 266
            and projected.get("changed_inputs") == ["src/shared_text_stage.inc"], "SEP-P projected manifest invalid")
    for relative, expected_hash in projected["inputs"].items():
        require(sha((CANDIDATE / relative).read_bytes()) == expected_hash, f"built source identity mismatch: {relative}")

    input_files = {
        path.relative_to(CANDIDATE).as_posix()
        for folder in ("src", "scripts", "assets", "tiled")
        for path in (CANDIDATE / folder).rglob("*")
        if path.is_file() and "__pycache__" not in path.parts and path.suffix != ".pyc"
    }
    require(input_files == set(projected["inputs"]), "final candidate source inventory differs from 266-file manifest")
    observed_wiki = {
        path.relative_to(CANDIDATE / "wiki").as_posix(): sha(path.read_bytes())
        for path in (CANDIDATE / "wiki").rglob("*") if path.is_file()
    }
    wiki_hash = sha(json.dumps(wiki["files"], sort_keys=True, separators=(",", ":")).encode("utf-8"))
    require(len(observed_wiki) == 379 and observed_wiki == wiki["files"]
            and wiki_hash == wiki["inventory_sha256"], "frozen wiki inventory changed")

    result_path, attempt_path = SETUP / "build-result.json", SETUP / "build-attempt.json"
    wrapper_path, log_path = SETUP / "build-once.py.txt", SETUP / "build-first.log"
    result, attempt = json.loads(result_path.read_text(encoding="utf-8")), json.loads(attempt_path.read_text(encoding="utf-8"))
    require(result.get("card") == "SEP-B" and result.get("attempt") == 1, "single SEP-B result missing")
    require(result.get("wrapper_sha256") == sha(wrapper_path.read_bytes()), "SEP-B result does not bind reviewed wrapper")
    require(result.get("command") == EXPECTED_COMMAND and result.get("profile") == PROFILE,
            "SEP-B command/profile differs from approved complete keyboard adaptive 6809 build")
    require(result.get("child_deadline_seconds") == 595 and result.get("kill_grace_seconds") == 2
            and result.get("whole_phase_deadline_seconds") == 600, "SEP-B deadline metadata differs from approval")
    require(result.get("returncode") == 0 and not result.get("timed_out")
            and result.get("outcome") == "BUILD RETURNED ZERO; SEP-P PENDING",
            "SEP-B did not record one fresh successful build")
    require(isinstance(result.get("elapsed_seconds"), (int, float)) and 0 <= result["elapsed_seconds"] <= 600,
            "SEP-B exceeded its whole-phase deadline")
    require(attempt.get("command") == EXPECTED_COMMAND and attempt.get("profile") == PROFILE
            and attempt.get("attempt") == 1 and attempt.get("wrapper_sha256") == result["wrapper_sha256"],
            "SEP-B attempt marker does not bind the one-shot command")
    require(attempt.get("result") == result["outcome"]
            and attempt.get("result_sha256") == sha(result_path.read_bytes()), "SEP-B result marker hash mismatch")
    require(result.get("log") == log_path.name and log_path.is_file() and log_path.stat().st_size > 0,
            "SEP-B build log is missing or empty")
    log_bytes = log_path.read_bytes()
    require(result.get("log_sha256") == sha(log_bytes), "SEP-B raw log hash mismatch")
    log_text = log_bytes.decode("utf-8", errors="replace")

    build = CANDIDATE / "build"
    receipt_path = build / "source-build-receipt.json"
    receipt_bytes = receipt_path.read_bytes()
    receipt = json.loads(receipt_bytes)
    require(sha(receipt_bytes) == result.get("receipt_sha256"), "SEP-B receipt hash mismatch")
    require(receipt.get("profile") == "complete" and receipt.get("state") == "complete"
            and receipt.get("inputs") == projected["inputs"] and len(receipt["inputs"]) == 266,
            "receipt does not bind the complete profile and projected 266-file source")
    config_path = build / "source-reference-adaptive.json"
    require(config_path.is_file() and config_path.name in receipt.get("artifacts", {}),
            "receipt-bound adaptive stock-provenance config is missing")
    config_bytes = config_path.read_bytes()
    require(receipt["artifacts"][config_path.name] == sha(config_bytes), "provenance config differs from receipt")
    stock = load_stock_provenance()
    validated = stock.validate(CANDIDATE, build, json.loads(config_bytes))
    require(validated.get("state") == "complete" and validated.get("profile") == "complete"
            and validated == receipt, "stock provenance validation differs from build receipt")

    rom_path = build / "ladybug.rom"
    rom_bytes = rom_path.read_bytes()
    require(len(rom_bytes) == 65536 and receipt.get("artifacts", {}).get("ladybug.rom") == sha(rom_bytes)
            and result.get("rom_sha256") == sha(rom_bytes), "delivered GMC ROM identity or 64 KiB format mismatch")
    runtime_path = build / "ladybug-runtime.rom"
    require(runtime_path.stat().st_size == 16384
            and receipt.get("artifacts", {}).get("ladybug-runtime.rom") == sha(runtime_path.read_bytes()),
            "separate padded resident runtime identity/16 KiB format mismatch")

    helper_path = build / "ladybug-highscore-helper.bin"
    helper, helper_map_bytes = helper_path.read_bytes(), (build / "ladybug-highscore-helper.map").read_bytes()
    require(receipt.get("artifacts", {}).get(helper_path.name) == sha(helper), "helper binary differs from receipt")
    require(receipt.get("artifacts", {}).get("ladybug-highscore-helper.map") == sha(helper_map_bytes),
            "helper map differs from receipt")
    helper_map = helper_map_bytes.decode("utf-8")
    rank_rows = tmx_rank_rows(CANDIDATE / "tiled/coco-high-score-screen.tmx")
    compiled = schedule_from_binary(helper, helper_map)
    assert_good_schedule(compiled, rank_rows)
    require(len(helper) == 3070 and len(helper) <= 3136, "compiled helper is not exactly 3,070 bytes within cap")

    known_bad = (FIXTURE / "ladybug-highscore-helper.bin").read_bytes()
    known_bad_map = (FIXTURE / "ladybug-highscore-helper.map").read_text(encoding="utf-8")
    require(sha(known_bad) == BASELINE["helper"]["sha256"]
            and sha((FIXTURE / "ladybug-highscore-helper.map").read_bytes()) == BASELINE["helper_map"]["sha256"],
            "pinned compiled negative-control fixture identity changed")
    bad_records = schedule_from_binary(known_bad, known_bad_map)
    require(len(bad_records) == 20 and sum(cell(item[2])[0] == 5 for item in bad_records) == 2,
            "known-bad helper no longer contains exactly two row-5 jobs")
    rejected_bad = False
    try:
        assert_good_schedule(bad_records, rank_rows)
    except AssertionError as error:
        rejected_bad = "separator row 5" in str(error)
    require(rejected_bad, "known-bad compiled helper was not rejected for row-5 writes")

    layout_map = (build / "ladybug.map").read_text(encoding="utf-8")
    resident_end, asset_start, asset_end = (symbol(layout_map, name) for name in ("resident_end", "asset_start", "asset_end"))
    resident_used, assets_used = resident_end - 0xC000, asset_end - asset_start
    presentation_bytes = (build / "ladybug-presentation-runtime.bin").stat().st_size
    capacity = [tuple(map(int, item)) for item in re.findall(
        r"^build: resident (\d+)/(\d+) bytes; assets (\d+)/(\d+) bytes$", log_text, re.M)]
    require(capacity and capacity[-1] == (resident_used, 8192, assets_used, 7680),
            "ordinary capacity summary differs from final map")
    require(all(r_limit == 8192 and a_limit == 7680 and resident <= 8160 and assets <= 7680
                for resident, r_limit, assets, a_limit in capacity), "ordinary build capacity guards failed")
    require(resident_used <= 8160 and 8192 - resident_used >= 32
            and asset_start == 0xE000 and assets_used <= 7680,
            "resident minimum headroom or asset cap failed")
    require(presentation_bytes == 1280, "presentation module differs from fixed 1,280-byte cap")
    require(time.monotonic() - STARTED <= LIMIT_SECONDS, "SEP-P exceeded its 60-second whole-phase deadline")
    print(json.dumps({
        "card": "SEP-P",
        "receipt_sha256": sha(receipt_bytes),
        "rom_sha256": sha(rom_bytes),
        "schedule_jobs": len(compiled),
        "separator_row5_jobs": 0,
        "baseline_negative_control": "20-job helper rejected for row-5 writes",
        "stock_provenance": "complete",
        "capacity_bytes": {
            "resident_used_of_8192": resident_used,
            "resident_free_minimum": 8192 - resident_used,
            "assets_used_of_7680": assets_used,
            "presentation_used_of_1280": presentation_bytes,
            "helper_used_of_3136": len(helper),
        },
        "duration_seconds": round(time.monotonic() - STARTED, 3),
    }, indent=2))


if __name__ == "__main__":
    main()
