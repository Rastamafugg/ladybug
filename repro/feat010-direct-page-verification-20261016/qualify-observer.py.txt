def require(ok,message):
    if not ok: raise AssertionError(message)
def sha(path): return hashlib.sha256(path.read_bytes()).hexdigest()
def guarded(tree,diagnostic):
    hits=[n for n in ast.walk(tree) if isinstance(n,ast.If) and len(n.body)==1
          and isinstance(n.body[0],ast.Raise) and any(isinstance(a,ast.Constant) and a.value==diagnostic for a in ast.walk(n.body[0]))]
    require(len(hits)==1,'unique assertion: '+diagnostic)
    return hits[0]
def root_node(tree):
    nodes=[n for n in tree.body if isinstance(n,ast.Assign) and any(isinstance(t,ast.Name) and t.id=='ROOT' for t in n.targets)]
    require(len(nodes)==1,'unique ROOT assignment')
    return nodes[0]
def check_root(node):
    expected=ast.parse('Path('+repr(bound_root)+')',mode='eval').body
    require(ast.dump(node.value)==ast.dump(expected),'external ROOT binding')
def rejects(node,namespace,diagnostic):
    module=ast.fix_missing_locations(ast.Module(body=[copy.deepcopy(node)],type_ignores=[]))
    try: exec(compile(module,'scoped-observer-assertion','exec'),namespace.copy())
    except SystemExit as exc:
        require(str(exc)==diagnostic,'unexpected diagnostic: '+str(exc))
        return True
    return False
def accepts(node,namespace,diagnostic):
    require(not rejects(node,namespace,diagnostic),'unexpected rejection: '+diagnostic)
def mask_test(tree,node):
    clone=copy.deepcopy(tree)
    target=next(n for n in ast.walk(clone) if isinstance(n,ast.If) and n.lineno==node.lineno and n.col_offset==node.col_offset)
    target.test=ast.Constant(value='APPROVED_PREDICATE_MASK')
    return ast.dump(clone)

    return b''.join(rows)
