"""Approved MC-M: one manual menu-continuity visit; close only the verified child."""
from pathlib import Path
import hashlib
import json
import os
import signal
import subprocess
import sys
import time

WORK = Path(__file__).resolve().parent
BASE = WORK / 'candidate'
EXE = Path('/mnt/e/projects/ladybug/docs/reference/xroar/src/xroar')
ROM_SHA = '2c8add7b35ca4b5b01c119ed9ad35b0e1bf6ba9a48c26bc43262ac0321084e9c'
RECEIPT_SHA = '6959493425fcca3789bee8dcb4219fb65fbd211002de72c4b92622b6be9836d4'
EXE_SHA = 'bad8de1a7a5cb1be5f55ff9bbb213442b403bc83c1f833c343523d5c318f6dbf'

def digest(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()

def verify():
    assert digest(BASE / 'build/ladybug.rom') == ROM_SHA
    assert digest(BASE / 'build/source-build-receipt.json') == RECEIPT_SHA
    assert digest(EXE) == EXE_SHA
    assert json.loads((WORK / 'post-relocation-result.json').read_bytes())['outcome'].startswith('PASS COMPILED SOURCE/FIT')
    for name in ('manual-attempt-mcm-20261018.json', 'manual-launch-mcm-20261018.json', 'manual-cleanup-mcm-20261018.json'):
        assert not (WORK / name).exists(), 'Attempt already exists: ' + name

def identity(pid):
    proc = Path('/proc') / str(pid)
    return {'pid': pid, 'exe': os.readlink(proc / 'exe'),
            'argv': proc.joinpath('cmdline').read_bytes().split(b'\0')[:-1],
            'start_ticks': proc.joinpath('stat').read_text().rsplit(')', 1)[1].split()[19]}

def guard(phase_start_ms):
    verify()
    assert int(time.time() * 1000) < phase_start_ms + 15000, 'Startup boundary expired'
    with (WORK / 'manual-attempt-mcm-20261018.json').open('x') as handle:
        json.dump({'card': 'MC-M', 'attempt': 1, 'phase_start_ms': phase_start_ms,
                   'limit_seconds': 60, 'cleanup_by_seconds': 57}, handle)
    command = [str(EXE), '-ui', 'gtk3', '-ao', 'pulse', '-machine', 'coco3',
               '-machine-cpu', '6809', '-ram', '512', '-ram-init', 'random',
               '-cart-type', 'gmc', '-cart-rom', str(BASE / 'build/ladybug.rom'),
               '-cart-autorun', '-tv-input', 'rgb', '-joy-right', '', '-joy-left', '']
    log = (WORK / 'manual-xroar-mcm-20261018.log').open('xb')
    process = subprocess.Popen(command, cwd=BASE, stdin=subprocess.DEVNULL,
                               stdout=log, stderr=subprocess.STDOUT, start_new_session=True)
    expected = None
    cleanup = {'card': 'MC-M', 'signals': [], 'scope': 'Only exact owned child'}
    try:
        expected = identity(process.pid)
        assert expected['exe'] == str(EXE.resolve())
        assert expected['argv'] == [os.fsencode(value) for value in command]
        record = {**expected, 'argv': command, 'rom_sha256': ROM_SHA,
                  'receipt_sha256': RECEIPT_SHA, 'guard_sha256': digest(Path(__file__)),
                  'phase_start_ms': phase_start_ms, 'launch_epoch_ms': int(time.time() * 1000),
                  'debugger_or_monitor': False, 'input': 'Manual Down/Return and Back; no automation',
                  'manual_result': 'Pending', 'limit_seconds': 60}
        (WORK / 'manual-launch-mcm-20261018.json').write_text(json.dumps(record, indent=2) + '\n')
        while int(time.time() * 1000) < phase_start_ms + 57000:
            if process.poll() is not None:
                break
            time.sleep(.05)
    finally:
        cleanup['cleanup_started_ms'] = int(time.time() * 1000)
        if process.poll() is None:
            try:
                same = expected is not None and identity(process.pid) == expected
            except (FileNotFoundError, ProcessLookupError):
                same = False
            if same:
                try:
                    process.send_signal(signal.SIGTERM)
                    cleanup['signals'].append('TERM')
                    process.wait(timeout=1)
                except subprocess.TimeoutExpired:
                    try:
                        same = identity(process.pid) == expected
                    except (FileNotFoundError, ProcessLookupError):
                        same = False
                    if same:
                        try:
                            process.send_signal(signal.SIGKILL)
                            cleanup['signals'].append('KILL')
                        except ProcessLookupError:
                            pass
                        try:
                            process.wait(timeout=1)
                        except subprocess.TimeoutExpired:
                            cleanup['kill_wait_expired'] = True
                except ProcessLookupError:
                    pass
            else:
                cleanup['identity_mismatch'] = True
        cleanup.update(owned_process_exited=process.poll() is not None,
                       returncode=process.poll(),
                       phase_elapsed_ms=int(time.time() * 1000) - phase_start_ms)
        cleanup['within_60_seconds'] = cleanup['phase_elapsed_ms'] <= 60000
        (WORK / 'manual-cleanup-mcm-20261018.json').write_text(json.dumps(cleanup, indent=2) + '\n')
        log.close()

def main():
    assert len(sys.argv) == 3 and sys.argv[1] in ('--start', '--guard')
    phase_start_ms = int(sys.argv[2])
    if sys.argv[1] == '--guard':
        guard(phase_start_ms)
        return
    verify()
    with (WORK / 'manual-driver-mcm-20261018.log').open('xb') as log:
        child = subprocess.Popen([sys.executable, '-B', __file__, '--guard', str(phase_start_ms)],
                                 stdin=subprocess.DEVNULL, stdout=log, stderr=subprocess.STDOUT,
                                 start_new_session=True)
    until = time.monotonic() + 5
    while time.monotonic() < until:
        record = WORK / 'manual-launch-mcm-20261018.json'
        if record.exists():
            print(record.read_text(), flush=True)
            return
        if child.poll() is not None:
            break
        time.sleep(.02)
    raise SystemExit('No verified manual launch; stop without retry')

if __name__ == '__main__':
    main()
